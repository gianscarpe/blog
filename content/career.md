document.addEventListener('DOMContentLoaded', function () {
  const citation = document.getElementById('career-citation');
  document.querySelectorAll('.career-viz svg g[fill="#898781"] > rect').forEach(function (marker) {
    const title = marker.querySelector('title');
    if (!title) return;
    const label = title.textContent;
    marker.setAttribute('tabindex', '0');
    marker.setAttribute('role', 'button');
    marker.setAttribute('aria-label', label);
    marker.style.cursor = 'pointer';
    const hitArea = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
    hitArea.setAttribute('x', String(Number(marker.getAttribute('x')) - 3));
    hitArea.setAttribute('y', '306');
    hitArea.setAttribute('width', '18');
    hitArea.setAttribute('height', '28');
    hitArea.setAttribute('fill', 'transparent');
    hitArea.setAttribute('stroke', 'none');
    marker.parentNode.insertBefore(hitArea, marker);
    hitArea.addEventListener('pointerenter', function () { citation.textContent = label; });
    hitArea.addEventListener('click', function () { citation.textContent = label; marker.focus(); });
    marker.addEventListener('pointerenter', function () { citation.textContent = label; });
    marker.addEventListener('focus', function () { citation.textContent = label; });
    marker.addEventListener('click', function () { citation.textContent = label; });
    marker.addEventListener('keydown', function (event) {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        citation.textContent = label;
      }
    });
  });
});
</script>
